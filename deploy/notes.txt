# ─── BÁT TỰ (bazi.sincely.io.vn) ────────────────────────────────
# Repo nằm ở ../Web-Bazi-Global (clone git riêng).
# Khối service ở trên do ../Web-Bazi-Global/deploy/install.sh đặt vào —
# chạy lại script đó để cập nhật, đừng sửa tay giữa 2 dòng mốc BAZI START/END.
#
# Env cần khai báo trong .env (cùng folder docker-compose.yml):
#   BAZI_DB_PASSWORD=<openssl rand -base64 32>
#   BAZI_AUTH_SECRET=<openssl rand -hex 48>
#   BAZI_CRON_SECRET=<openssl rand -hex 24>
#   BAZI_ANTHROPIC_API_KEY=...   (optional — thiếu thì luận giải dùng bản mộc)
#   BAZI_RESEND_API_KEY=...      (optional — thiếu thì không gửi email)
#   BAZI_GOOGLE_ID=...           (optional — thiếu thì ẩn nút Google)
#   BAZI_GOOGLE_SECRET=...
#
# SSL cert lần đầu (chưa có cert thì nginx không start được với bazi.conf, nên
# xin cert TRƯỚC rồi mới copy file conf vào — copy sớm là sập mọi domain khác):
#   docker compose up -d --build bazi-db bazi-migrate bazi bazi-cron
#   docker compose run --rm certbot certonly --webroot -w /var/www/certbot \
#     -d bazi.sincely.io.vn
#   cp ../Web-Bazi-Global/deploy/nginx/bazi.conf nginx/conf.d/bazi.conf
#   docker compose exec nginx nginx -t && docker compose exec nginx nginx -s reload
#
# Deploy / cập nhật code (sau khi git pull repo Web-Bazi-Global):
#   docker compose up -d --build bazi-migrate bazi
#
# Lưu ý: PHẢI build bazi-migrate cùng lúc. Nó chạy migration rồi thoát, bazi
#         đợi nó qua service_completed_successfully. migrate deploy chỉ chạy
#         xuôi theo các file migration đã commit, không tự sinh và không reset.
#
# Backup: service `backup` chỉ backup SQL Server, KHÔNG bao gồm Postgres này.
#   docker compose exec -T bazi-db pg_dump -U bazi bazi | gzip > bazi-$(date +%F).sql.gz
#
# Check dung lượng:
#   docker system df -v | grep bazi
